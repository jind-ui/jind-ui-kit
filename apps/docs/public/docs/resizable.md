# Resizable

**Category:** layout

## Import

```tsx
import { Resizable, ResizablePanel, ResizableHandle } from 'jind-ui-kit';
import type { ResizableProps, ResizablePanelProps, ResizableHandleProps } from 'jind-ui-kit';
```
